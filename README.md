const GITHUB_USERNAME = "TharukiJ";


/*
    Get all PUBLIC repositories
*/

async function getGitHubRepositories() {

    let repositories = [];

    let page = 1;

    while (true) {

        const response = await fetch(
            `https://api.github.com/users/${GITHUB_USERNAME}/repos?per_page=100&page=${page}&type=owner`
        );

        if (!response.ok) {
            throw new Error("Unable to load GitHub repositories.");
        }

        const data = await response.json();

        if (data.length === 0) {
            break;
        }

        repositories = repositories.concat(data);

        if (data.length < 100) {
            break;
        }

        page++;
    }

    return repositories;
}


/*
    Get language statistics for one repository
*/

async function getRepositoryLanguages(owner, repo) {

    const response = await fetch(
        `https://api.github.com/repos/${owner}/${repo}/languages`
    );

    if (!response.ok) {
        return {};
    }

    return await response.json();
}


/*
    Calculate total language bytes
*/

async function calculateGitHubLanguages() {

    const repositories = await getGitHubRepositories();

    const languageTotals = {};

    /*
        Get language information from every repository
    */

    for (const repo of repositories) {

        /*
            Skip forks so your own work is represented
        */

        if (repo.fork) {
            continue;
        }

        const languages =
            await getRepositoryLanguages(
                GITHUB_USERNAME,
                repo.name
            );


        for (const [language, bytes] of Object.entries(languages)) {

            if (!languageTotals[language]) {
                languageTotals[language] = 0;
            }

            languageTotals[language] += bytes;
        }
    }


    /*
        Convert bytes into percentages
    */

    const totalBytes =
        Object.values(languageTotals)
        .reduce(
            (total, value) => total + value,
            0
        );


    const languagePercentages =
        Object.entries(languageTotals)

        .map(([language, bytes]) => ({

            language: language,

            percentage:
                (bytes / totalBytes) * 100

        }))

        .sort(
            (a, b) =>
                b.percentage - a.percentage
        );


    return languagePercentages;
}


/*
    Create GitHub Radar Chart
*/

async function createGitHubChart() {

    const loading =
        document.getElementById("githubLoading");

    const error =
        document.getElementById("githubError");


    try {

        const languages =
            await calculateGitHubLanguages();


        if (!languages.length) {

            throw new Error(
                "No public language data found."
            );

        }


        /*
            Show top 7 languages.

            This keeps the radar chart clean.
        */

            languages.slice(0, 7);


        const labels =
            topLanguages.map(
                item => item.language
            );


        const values =
            topLanguages.map(
                item =>

                        item.percentage.toFixed(1)
                    )
            );


        /*
            Create chart
        */

        new Chart(

            document.getElementById("githubChart"),

            {

                type: "radar",

                data: {

                    datasets: [

                        {

                            borderColor: "#79d45b",

                            backgroundColor:
                                "rgba(83, 180, 75, 0.25)",

                            pointBackgroundColor:
                                "#79d45b",

                            pointBorderColor:
                                "#79d45b",

                            pointHoverBackgroundColor:
                                "#ffffff",

                            pointHoverBorderColor:
                                "#79d45b",

                            fill: true
                        }

                    ]
                },


                options: {

                    responsive: true,

                    maintainAspectRatio: false,


                    scales: {

                        r: {

                            min: 0,

                            max: 100,

                            beginAtZero: true,


                            ticks: {

                                display: false,

                                stepSize: 20
                            },


                            grid: {

                                color:
                                    "rgba(139, 148, 158, 0.22)"
                            },


                            angleLines: {

                                color:
                                    "rgba(139, 148, 158, 0.22)"
                            },


                            pointLabels: {

                                color: "#ffffff",

                                font: {
                                    size: 10
                                },

                                callback: function(value) {

                                    const item =
                                        topLanguages.find(
                                            x =>
                                                x.language === value
                                        );

                                    if (item) {

                                        return [
                                            value,
                                            item.percentage

                                        ];

                                    }

                                    return value;
                                }
                            }
                        }
